등급 값은 'VIP', '일반', '휴면' 셋 중 하나야.

-- ① VIP 고객은 몇 명이고, 누적구매액 합계는 얼마인가요?
-- ② 인스타그램으로 가입한 고객의 이름은?

-- ① VIP 고객 수와 누적구매액 합계
SELECT COUNT(*) AS VIP_고객수,
       SUM(누적구매액) AS VIP_누적구매액_합계
FROM 고객명단
WHERE 등급 = 'VIP';

-- ② 인스타그램으로 가입한 고객의 이름
SELECT 이름
FROM 고객명단
WHERE 가입경로 = '인스타그램';

-- ③ 미처리 문의를 담당자별로 집계
SELECT 담당자,
       COUNT(*) AS 미처리_문의수
FROM 문의접수
WHERE 상태 = '미처리'
GROUP BY 담당자;

-------------------------------------------------------------------------------------------------------
## cf.-- 여기서 미처리_문의수의 숫자가 높고, 이름이 ㄱ-ㅎ 순서로 나오게 정리하고 싶다면, quary를 어떻게 수정하는게 좋아?
여기에 **정렬 기준을 정해 주는 친구가 ORDER BY**예요.

당신이 원하는 건

먼저 미처리_문의수를 내림차순(큰 숫자 → 작은 숫자) 으로,
개수가 같다면 담당자를 가나다(ㄱ-ㅎ) 순으로
정렬하는 거죠.

----------------------

SELECT 담당자,
       COUNT(*) AS 미처리_문의수
FROM 문의접수
WHERE 상태 = '미처리'
GROUP BY 담당자

ORDER BY 미처리_문의수 DESC, 담당자 ASC;

-- ④ 매출이 가장 높았던 날과 그날 매출
SELECT 날짜,
       매출
FROM 일일지표
ORDER BY 매출 DESC
LIMIT 1;

-- ⑤ 이벤트 응모자 중 휴면 고객의 이름과 관심분야
SELECT e.이름,
       e.관심분야
FROM 이벤트응모 e
JOIN 고객명단 c ON e.이메일 = c.이메일
WHERE c.등급 = '휴면';
